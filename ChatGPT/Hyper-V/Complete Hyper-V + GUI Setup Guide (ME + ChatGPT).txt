## verified Windows host Has Enable the Hyper-V
# Verify Hyper-V is available
Get-WindowsOptionalFeature -Online -FeatureName Microsoft-Hyper-V-All

# Enable if not present (requires reboot)
Enable-WindowsOptionalFeature -Online -FeatureName Microsoft-Hyper-V -All

# Verify you are on Windows 10 build 14393+ (required for AF_HYPERV sockets)
[System.Environment]::OSVersion.Version



## Enabel The Hyper-V
dism.exe /Online /Enable-Feature /All /FeatureName:Microsoft-Hyper-V



## Build the Alpine Linux VM image
# Create build VM (temporary, used to install Alpine)
$buildVM = New-VM -Name "IsolatedBrowser-Build" `
    -MemoryStartupBytes 2GB `
    -Generation 2 `
    -NewVHDPath "C:\VMs\IsolatedBrowser.vhdx" `
    -NewVHDSizeBytes 10GB

# Set CPU (Program Capacity)
Set-VMProcessor $buildVM.Name -Count 2
# Attach ISO File (for OS installation)
Set-VMDvdDrive -VMName $buildVM.Name `
    -Path "C:\ISOs\alpine-extended-3.23.3-x86_64.iso"

# Disable Secure Boot for Alpine Gen2
Set-VMFirmware -VMName $buildVM.Name `
    -EnableSecureBoot Off
# Configure Boot Order
Set-VMFirmware -VMName $buildVM.Name `
    -FirstBootDevice (Get-VMDvdDrive -VMName $buildVM.Name)
Set-VMFirmware -VMName $buildVM.Name `
    -SecureBootTemplate OpenSourceShieldedVM

# Start the VM
Start-VM $buildVM.Name
# Open VM GUI
vmconnect.exe localhost $buildVM.Name


# # Stop the VM
Stop-VM $buildVM.Name -Force
